cmake_minimum_required(VERSION 3.14)
project(minigpu_source_scope_cpu LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_executable(minigpu_source_scope_cpu
  "../../src/minigpu_source_scope.cpp" "../../src/minigpu_capture_cleanup_owner.cpp"
  "source_scope_cpu_test.cpp")
target_compile_definitions(minigpu_source_scope_cpu PRIVATE MINIGPU_SOURCE_METADATA_TEST)
target_compile_features(minigpu_source_scope_cpu PRIVATE cxx_std_17)
add_executable(minigpu_setup_layout_cpu "setup_layout_cpu_test.cpp"
  "../../src/minigpu_source_scope.cpp" "../../src/minigpu_capture_source_owner.cpp")
target_compile_features(minigpu_setup_layout_cpu PRIVATE cxx_std_17)
target_link_libraries(minigpu_setup_layout_cpu PRIVATE Threads::Threads)
option(MINIGPU_TEST_KNOWN_ALLOCATION_LAYOUT "Require the qualified Release C++20 layout witness" OFF)
if(MINIGPU_TEST_KNOWN_ALLOCATION_LAYOUT)
  target_compile_features(minigpu_source_scope_cpu PRIVATE cxx_std_20)
  target_compile_definitions(minigpu_source_scope_cpu PRIVATE MINIGPU_EXPECT_KNOWN_ALLOCATION_LAYOUT)
  target_compile_features(minigpu_setup_layout_cpu PRIVATE cxx_std_20)
  target_compile_definitions(minigpu_setup_layout_cpu PRIVATE MINIGPU_EXPECT_KNOWN_ALLOCATION_LAYOUT)
endif()
target_link_libraries(minigpu_source_scope_cpu PRIVATE Threads::Threads)
add_executable(minigpu_allocation_layout_unsupported_cpu "allocation_layout_unsupported_cpu_test.cpp")
target_compile_features(minigpu_allocation_layout_unsupported_cpu PRIVATE cxx_std_17)
target_compile_definitions(minigpu_allocation_layout_unsupported_cpu PRIVATE MINIGPU_ALLOCATION_LAYOUT_FORCE_UNSUPPORTED)
if(MSVC)
  target_compile_options(minigpu_source_scope_cpu PRIVATE /W4 /WX /EHsc)
  target_compile_options(minigpu_allocation_layout_unsupported_cpu PRIVATE /W4 /WX /EHsc)
  target_compile_options(minigpu_setup_layout_cpu PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(minigpu_source_scope_cpu PRIVATE -Wall -Wextra -Werror)
  target_compile_options(minigpu_allocation_layout_unsupported_cpu PRIVATE -Wall -Wextra -Werror)
  target_compile_options(minigpu_setup_layout_cpu PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(minigpu_funded_setup_cpu "funded_setup_cpu_test.cpp"
  "../../src/minigpu_source_scope.cpp" "../../src/minigpu_capture_source_owner.cpp")
target_compile_features(minigpu_funded_setup_cpu PRIVATE cxx_std_17)
target_link_libraries(minigpu_funded_setup_cpu PRIVATE Threads::Threads)
if(MINIGPU_TEST_KNOWN_ALLOCATION_LAYOUT)
  target_compile_features(minigpu_funded_setup_cpu PRIVATE cxx_std_20)
endif()
if(MSVC)
  target_compile_options(minigpu_funded_setup_cpu PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(minigpu_funded_setup_cpu PRIVATE -Wall -Wextra -Werror)
endif()

# Test-only C ABI bridge DLL; no GPU/native context library or execution.
add_library(minigpu_setup_bridge_cpu SHARED "setup_bridge_cpu_test.cpp" "setup_custody_abi_c.c"
  "../../src/minigpu_source_scope.cpp" "../../src/minigpu_capture_source_owner.cpp")
target_compile_features(minigpu_setup_bridge_cpu PRIVATE cxx_std_17 c_std_11)
target_link_libraries(minigpu_setup_bridge_cpu PRIVATE Threads::Threads)
if(MINIGPU_TEST_KNOWN_ALLOCATION_LAYOUT)
  target_compile_features(minigpu_setup_bridge_cpu PRIVATE cxx_std_20)
endif()
if(MSVC)
  target_compile_options(minigpu_setup_bridge_cpu PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(minigpu_setup_bridge_cpu PRIVATE -Wall -Wextra -Werror)
endif()
